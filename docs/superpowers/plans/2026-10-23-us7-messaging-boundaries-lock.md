```python
from __future__ import annotations

from pathlib import Path

from app.compliance import STANDARD_DISCLAIMERS
from app.product_fields_extract import allowed_product_url

_REPO_ROOT = Path(__file__).resolve().parents[2]
_PRD = _REPO_ROOT / "docs" / "2026-10-22-PRD-保额保费摘录日期.md"

_PITCH = "第一次会见拿出 3 个方案"

_EXPECTED_DISCLAIMERS = {
    "zh-CN": (
        "本内容由 AI 根据用户提供的信息和公开资料生成，仅供一般信息参考，不构成持牌财务、保险、税务、法律或投资建议。"
        "保险产品条款、费用、收益、保障范围和适用性应以保险公司正式文件及持牌专业人士意见为准。"
        "请在作出任何购买或配置决定前，咨询合资格专业顾问。"
    ),
    "zh-TW": (
        "本內容由 AI 根據用戶提供的資訊和公開資料生成，僅供一般資訊參考，不構成持牌財務、保險、稅務、法律或投資建議。"
        "保險產品條款、費用、收益、保障範圍和適用性應以保險公司正式文件及持牌專業人士意見為準。"
        "請在作出任何購買或配置決定前，諮詢合資格專業顧問。"
    ),
    "en": (
        "This content is generated by AI from information you provided and publicly available "
        "materials. It is for general information only and does not constitute licensed financial, "
        "insurance, tax, legal, or investment advice. Product terms, fees, benefits, coverage, and "
        "suitability are subject to the insurer's official documents and advice from licensed "
        "professionals. Please consult a qualified adviser before any purchase or configuration decision."
    ),
}


def test_us7_pitch_unchanged_in_prd() -> None:
    text = _PRD.read_text(encoding="utf-8")
    assert _PITCH in text


def test_us7_standard_disclaimers_unchanged() -> None:
    for loc, expected in _EXPECTED_DISCLAIMERS.items():
        assert STANDARD_DISCLAIMERS[loc] == expected


def test_us7_allowed_product_url_china_life_only() -> None:
    assert allowed_product_url(
        "https://www.chinalife.com.hk/zh-hk/products/insurance/medical/example-plan"
    )
    assert not allowed_product_url("https://example.com/zh-hk/products/insurance/medical/x")
    assert not allowed_product_url("https://www.chinalife.com.hk/zh-hk/about")
    assert not allowed_product_url(
        "https://www.chinalife.com.hk/zh-hk/products/insurance"
    )


def test_us7_zh_cn_disclaimer_non_advice_clause() -> None:
    assert "不构成持牌财务、保险、税务、法律或投资建议" in STANDARD_DISCLAIMERS["zh-CN"]
```
